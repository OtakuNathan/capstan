    include(CheckIPOSupported)
    check_ipo_supported(RESULT capstan_ipo_supported OUTPUT capstan_ipo_error LANGUAGES CXX)
    if(NOT capstan_ipo_supported)
        message(FATAL_ERROR "LTO unavailable: ${capstan_ipo_error}")
    endif()
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION ON)

add_library(capstan INTERFACE)

target_link_libraries(capstan INTERFACE lease::lease dynabridge::dynabridge flux_foundry::flux_foundry Threads::Threads)
if(CAPSTAN_NO_EXCEPTIONS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "CAPSTAN_NO_EXCEPTIONS currently requires GCC/Clang")
    endif()
    target_compile_options(capstan INTERFACE -fno-exceptions)
    target_compile_definitions(capstan INTERFACE
        CAPSTAN_NO_EXCEPTION_STRICT=$<BOOL:${CAPSTAN_NO_EXCEPTION_STRICT}>
        FLUX_FOUNDRY_NO_EXCEPTION_STRICT=$<BOOL:${CAPSTAN_NO_EXCEPTION_STRICT}>
        DYNABRIDGE_NO_EXCEPTION_STRICT=$<BOOL:${CAPSTAN_NO_EXCEPTION_STRICT}>)
endif()
if(CAPSTAN_NATIVE_ARCH)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "CAPSTAN_NATIVE_ARCH currently supports GCC/Clang -march=native")
    endif()
    target_compile_options(capstan INTERFACE -march=native)
endif()
if(CAPSTAN_VECTORIZATION_REPORT)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        message(FATAL_ERROR "CAPSTAN_VECTORIZATION_REPORT currently requires GCC")
    endif()
    target_compile_options(capstan INTERFACE -fopt-info-vec-optimized)
endif()
option(CAPSTAN_FAST_RUNNER "Use FF fast runner without runner cancellation (experiment)" OFF)
if(CAPSTAN_FAST_RUNNER)
    target_compile_definitions(capstan INTERFACE CAPSTAN_FAST_RUNNER=1)
endif()
option(CAPSTAN_ASAN "Enable AddressSanitizer and UBSan" OFF)
option(CAPSTAN_TSAN "Enable ThreadSanitizer (separate build)" OFF)
if(CAPSTAN_ASAN AND CAPSTAN_TSAN)
    message(FATAL_ERROR "Use separate ASan and TSan builds")
endif()
if(CAPSTAN_ASAN)
    target_compile_options(capstan INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    target_link_options(capstan INTERFACE -fsanitize=address,undefined -no-pie)
elseif(CAPSTAN_TSAN)
    target_compile_options(capstan INTERFACE -fsanitize=thread -fno-omit-frame-pointer -fno-pie)
    target_link_options(capstan INTERFACE -fsanitize=thread -no-pie)
endif()
function(capstan_test name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE capstan)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endfunction()
