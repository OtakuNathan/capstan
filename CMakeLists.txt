    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")

    endif()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
        set(capstan_native_flag -mcpu=native)
    else()
        set(capstan_native_flag -march=native)
    endif()
    include(CheckCXXCompilerFlag)
    check_cxx_compiler_flag("${capstan_native_flag}" capstan_native_supported)
    if(NOT capstan_native_supported)
        message(FATAL_ERROR "Compiler does not support ${capstan_native_flag}; disable CAPSTAN_NATIVE_ARCH")
    endif()
    target_compile_options(capstan INTERFACE "${capstan_native_flag}")
endif()

function(capstan_test name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE capstan)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endfunction()
